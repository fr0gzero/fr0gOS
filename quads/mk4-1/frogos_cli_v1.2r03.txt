# FrogOS v1.2 r03 - MK41 (BETAFPV Matrix 1S 5-in-1 II, BETAFPVG473_V2, analog)
# Paste into the Betaflight CLI after flashing frogos_v1.2r03_mk4-1_BETAFPVG473_V2.hex
# with "Full chip erase" ticked. Restores Tom's setup plus the FrogOS layout.
# Analog NTSC: 30 columns x 13 rows. Column 0 = left-justified, column 29 = right-justified.

batch start
defaults nosave
board_name BETAFPVG473_V2
manufacturer_id BEFH

# features
feature -AIRMODE
feature OSD

# beeper
beacon RX_LOST
beacon RX_SET

# modes
aux 0 0 0 1700 2100 0 0
aux 1 1 1 900 1300 0 0
aux 2 35 3 1750 2100 0 0
# air mode on AUX 8 (Pocket left trim virtual switch)
aux 3 28 7 1700 2100 0 0

# vtx table (Matrix 1S 5-in-1 II: 25/100/200/400 mW + PIT)
vtxtable bands 6
vtxtable channels 8
vtxtable band 1 BOSCAM_A A FACTORY 5865 5845 5825 5805 5785 5765 5745 5725
vtxtable band 2 BOSCAM_B B FACTORY 5733 5752 5771 5790 5809 5828 5847 5866
vtxtable band 3 BOSCAM_E E FACTORY 5705 5685 5665 5645 5885 5905 5925 5945
vtxtable band 4 FATSHARK F FACTORY 5740 5760 5780 5800 5820 5840 5860 5880
vtxtable band 5 RACEBAND R FACTORY 5658 5695 5732 5769 5806 5843 5880 5917
vtxtable band 6 LOWRACE L FACTORY 5362 5399 5436 5473 5510 5547 5584 5621
vtxtable powerlevels 5
vtxtable powervalues 1 2 3 4 0
vtxtable powerlabels 25 100 200 400 PIT

# hardware
set acc_calibration = -23,35,26,1
set dshot_bidir = ON
set dshot_bitbang = AUTO
set motor_pwm_protocol = DSHOT300
set motor_poles = 12
set align_board_yaw = -135
set vbat_scale = 113

# vtx: Raceband 8, 200 mW
set vtx_band = 5
set vtx_channel = 8
set vtx_power = 3
set vtx_freq = 5917
set vcd_video_system = NTSC

# names: FROG0 top left, craft name top right and on the boot screen
set craft_name = MK41
set pilot_name = FR0G

# alarms
set osd_rssi_alarm = 0
set osd_rssi_dbm_alarm = -99
set osd_cap_alarm = 0
set stats_min_armed_time_s = 15
# 20 fps OSD (MAX7456 over SPI has plenty of headroom)
set osd_framerate_hz = 20
# battery noise in the outer 4 columns (SAG, LOW, LAND, DEAD; mono on analog)
set osd_frog_noise_pos = 2048
set osd_alt_alarm = 0

# ---- FrogOS analog layout: 30x13, x=0 left-justified, x=29 right-justified ----
# top bar: FR0G + blinking _ | craft name (no frog glyph on analog)
set osd_frog_tag_pos = 2048
set osd_frog_craft_pos = 2077
set osd_pilot_name_pos = 0
set osd_craft_name_pos = 0
# power, top left
set osd_vbat_pos = 2080
set osd_frog_batt_pos = 2112

# link, middle left
set osd_link_quality_pos = 2176

# camera / video format, bottom left

set osd_custom_msg1_pos = 2368
set osd_custom_msg2_pos = 2400
# time, top right
set osd_tim_2_pos = 2109

# state, middle right
set osd_disarmed_pos = 2205
set osd_flymode_pos = 2237
