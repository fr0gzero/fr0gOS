# fr0gOS 2.6 (analog, corners layout) - MK4-1
# 1. Flash fr0gOS_2.6_4-1_BETAFPVG473_V2.hex with Full Chip Erase ticked (v1.3 settings don't carry over)
# 2. Paste this whole file into the CLI. It saves and reboots.
# 3. Calibrate the accelerometer on the Setup tab (quad flat).

vtxtable band 6 LOWRACE L FACTORY 5362 5399 5436 5473 5510 5547 5584 5621
vtxtable powerlevels 5
vtxtable powervalues 1 2 3 4 0
vtxtable powerlabels 25 100 200 400 PIT

set dshot_bitbang = AUTO
set motor_pwm_protocol = DSHOT300
set motor_poles = 12
