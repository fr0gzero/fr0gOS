# fr0gOS fleet setup for Mark 6-1 (Angel 35, TBS Lucid FC + Lucid 60A AM32, analog)
# Betaflight 4.5.1, board TBS_LUCID_FC. Paste into the CLI with a battery plugged in, then the last line saves and reboots.

# --- receiver: ELRS (CRSF) on UART5 (R5/T5 pads) - board default, set explicitly
serial UART5 64 115200 57600 0 115200
set serialrx_provider = CRSF

# --- motors: AM32 ESC, bidirectional DShot for RPM filtering and RPM telemetry
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set motor_poles = 14

# --- analog OSD on the FC's own OSD chip (status showed MSP)
set osd_displayport_device = MAX7456
set vcd_video_system = AUTO

# --- VTX: SmartAudio on UART8 (TX8 pad), generic SmartAudio 2.1 table, Raceband 8 (5917 MHz), 200 mW
# If the VTX is IRC Tramp instead, change 2048 to 8192 on the next line and the power values to 25 100 200 400.
serial UART8 2048 115200 57600 0 115200
vtxtable bands 5
vtxtable channels 8
vtxtable band 1 BOSCAM_A A FACTORY 5865 5845 5825 5805 5785 5765 5745 5725
vtxtable band 2 BOSCAM_B B FACTORY 5733 5752 5771 5790 5809 5828 5847 5866
vtxtable band 3 BOSCAM_E E FACTORY 5705 5685 5665 5645 5885 5905 5925 5945
vtxtable band 4 FATSHARK F FACTORY 5740 5760 5780 5800 5820 5840 5860 5880
vtxtable band 5 RACEBAND R FACTORY 5658 5695 5732 5769 5806 5843 5880 5917
vtxtable powerlevels 4
vtxtable powervalues 14 20 23 26
vtxtable powerlabels 25 100 200 400
set vtx_band = 5
set vtx_channel = 8
set vtx_power = 3

# --- switches: same as 4-2 so the Pocket behaves identically (arm AUX1, angle, turtle, air mode on AUX8 / left trim)
aux 0 0 0 1800 2100 0 0
aux 1 1 1 900 1150 0 0
aux 2 6 2 1625 2100 0 0
aux 3 7 2 1300 1700 0 0
aux 4 35 3 1675 2100 0 0
aux 5 28 7 1700 2100 0 0
aux 6 40 4 1700 2100 0 0

# --- rates: Tom's preset (Betaflight rates)
rateprofile 0
set rates_type = BETAFLIGHT
set roll_rc_rate = 65
set pitch_rc_rate = 65
set yaw_rc_rate = 55
set roll_srate = 74
set pitch_srate = 74
set yaw_srate = 74
set roll_expo = 10
set pitch_expo = 10
set yaw_expo = 10
set thr_mid = 50
set thr_expo = 0

# --- names
set craft_name = MK6-1
set pilot_name = fr0gzero

save
